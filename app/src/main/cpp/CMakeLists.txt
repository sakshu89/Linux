cmake_minimum_required(VERSION 3.22.1)
project(linux_native)

set(LLAMA_CPP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/llama.cpp")
set(CACTUS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/cactus")

if(EXISTS "${LLAMA_CPP_DIR}/CMakeLists.txt")
    set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    add_subdirectory("${LLAMA_CPP_DIR}" llama EXCLUDE_FROM_ALL)
    add_library(linux_llama SHARED linux_llama_jni.cpp)
    target_include_directories(linux_llama PRIVATE "${LLAMA_CPP_DIR}/include")
    target_link_libraries(linux_llama PRIVATE llama log android)
else()
    add_library(linux_llama SHARED linux_llama_stub.cpp)
endif()

if(EXISTS "${CACTUS_DIR}/include/cactus_engine.h")
    add_library(linux_cactus SHARED linux_cactus_jni.cpp)
    target_include_directories(linux_cactus PRIVATE "${CACTUS_DIR}/include")
    if(EXISTS "${CACTUS_DIR}/lib/arm64-v8a/libcactus.so")
        add_library(cactus_engine SHARED IMPORTED)
        set_target_properties(cactus_engine PROPERTIES IMPORTED_LOCATION "${CACTUS_DIR}/lib/arm64-v8a/libcactus.so")
        target_link_libraries(linux_cactus PRIVATE cactus_engine log android)
    endif()
else()
    add_library(linux_cactus SHARED linux_cactus_stub.cpp)
endif()
